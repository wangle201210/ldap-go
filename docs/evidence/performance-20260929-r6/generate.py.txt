def require(condition, message):
    if not condition:
        raise ValueError(message)


def sha(data):
    return hashlib.sha256(data).hexdigest()


def text_file(path):
    require(path.is_file() and not path.is_symlink(), f'Missing/nonregular file: {path}')
    require(path.stat().st_size <= MAX_TEXT_BYTES, f'Oversize text: {path}; do not truncate')
    data = path.read_bytes()
    text = data.decode('utf-8')
    require('\x00' not in text, f'Binary content: {path}')
    # Scripts already use environment credentials; refuse new literal assignments.
    for match in re.finditer(r'\b[A-Z_]*PASSWORD\s*=\s*([^\s]+)', text):
        require('$' in match[1], f'Literal password assignment in {path}')
    require(not re.search(r'(?im)^\s*(?:rootpw|userPassword)\s*[:= ]', text),
            f'Credential-bearing content in {path}')
    require(not re.search(r'(?i)"(?:password|root_password|user_password)"\s*:', text),
            f'Credential-bearing JSON in {path}')
    require(not re.search(r'(?<!\S)-w\s+(?!["\']?\$)\S+', text),
            f'Literal command-line password in {path}')
    return data


def tsv(columns, rows):
    output = io.StringIO()
    writer = csv.writer(output, delimiter='\t', lineterminator='\n')
    writer.writerow(columns)
    writer.writerows(rows)
    return output.getvalue().encode()


def finite_positive(value):
    return isinstance(value, (int, float)) and math.isfinite(value) and value > 0


def inspect_report(path, expected, repeats, operations, root_dn=None):
    report = json.loads(text_file(path))
    cfg = report['config']
    require(not report.get('error'), f'{path}: report error')
    require(sorted(report['cleanup_done']) == sorted(ENDPOINTS), f'{path}: cleanup')
    require(sorted(e['name'] for e in cfg['endpoints']) == sorted(ENDPOINTS), f'{path}: endpoints')
    require(cfg['n'] == operations and cfg['repeats'] == repeats, f'{path}: batch dimensions')
    require(cfg['entries'] == 100000 and cfg['setup_disposable'], f'{path}: fixture')
    require(report['timeout_ms'] == 30000, f'{path}: timeout')
    if root_dn:
        require(cfg['root_bind_dn'] == root_dn, f'{path}: client root DN')
    groups = defaultdict(list)
    for sample in report['samples']:
        key = (sample['stage'], sample['method'], sample.get('members', 0), sample['endpoint'])
        require(not sample.get('error'), f'{path}: sample error')
        require(sample['completed'] == sample['operations'] == operations, f'{path}: completed count')
        require(len(sample['latency_ms']) == operations, f'{path}: latency count')
        require(all(finite_positive(t) for t in sample['latency_ms']), f'{path}: invalid latency')
        require(finite_positive(sample['total_ms']), f'{path}: invalid total')
        require(math.isclose(sum(sample['latency_ms']), sample['total_ms'], rel_tol=1e-9, abs_tol=1e-6),
                f'{path}: total differs from latency sum')
        requests = sample['requests']
        require(requests >= operations if sample['stage'] == 'nestedMembership' else requests == operations,
                f'{path}: request count')
        verification = (3 * operations if sample['stage'] == 'userBindWrong' else requests) + 2
        require(sample['verification_requests'] == verification, f'{path}: verification count')
        if root_dn:
            require(sample.get('root_bind_dn') == root_dn, f'{path}: sample root DN')
        groups[key].append(sample)
    require(set(groups) == {(*r, endpoint) for r in expected for endpoint in ENDPOINTS},
            f'{path}: missing or unexpected stage/method/member/endpoint group')
    for samples in groups.values():
        require(sorted(s['repeat'] for s in samples) == list(range(1, repeats + 1)), f'{path}: repeats')
    for stage, method, members in expected:
        counts = {(s['operations'], s['requests'], s['verification_requests'])
                  for endpoint in ENDPOINTS for s in groups[(stage, method, members, endpoint)]}
        require(len(counts) == 1, f'{path}: counts differ between endpoints/repeats')
    return report, groups


def common_table(run, groups):
    lines = [f'## {run}', '',
             '| Workload | Typical use | Calls | Before | Current | OpenLDAP | Relative | Time reduction |',
             '| --- | --- | ---: | ---: | ---: | ---: | ---: | ---: |']
    for batch, stage, method, members, label, frequency in ROWS:
        key = (run, batch, stage, method, members)
        before, current, native = [median(s['total_ms'] for s in groups[(*key, ep)]) for ep in ENDPOINTS]
        calls = f"{groups[(*key, 'before')][0]['operations']:,}"
        if stage == 'nestedMembership':
            calls += ' traversals'
        lines.append(f'| {label} | {frequency} | {calls} | {before:.2f} ms | {current:.2f} ms | '
                     f'{native:.2f} ms | {100*native/current:.1f}% | {100*(1-current/before):.1f}% |')
    return '\n'.join(lines)


def benchmark_rows(path, side):
    text = text_file(path).decode()
    require(re.search(r'^PASS$', text, re.M) and re.search(r'^ok\s', text, re.M),
            f'{path}: missing completed benchmark footer')
    rows = []
    repetitions = defaultdict(int)
    for line in text.splitlines():
        if not line.startswith('Benchmark'):
            continue
        parts = line.split()
        require(len(parts) >= 8 and len(parts) % 2 == 0, f'{path}: incomplete benchmark line')
        name = parts[0]
        metrics = dict(zip(parts[3::2], parts[2::2]))
        require({'ns/op', 'B/op', 'allocs/op'} <= metrics.keys(), f'{path}: missing metrics')
        repetitions[name] += 1
        for unit, value in metrics.items():
            number = float(value)
            require(math.isfinite(number) and number >= 0, f'{path}: invalid metric')
            rows.append([side, name, repetitions[name], int(parts[1]), unit, number])
    require(rows, f'{path}: no benchmark samples')
    return rows
