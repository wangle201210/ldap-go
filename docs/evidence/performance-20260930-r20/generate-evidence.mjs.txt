function remember(f) {
  const s = fs.lstatSync(f), previous = trackedReads.get(f);
  if (!s.isFile() || s.isSymbolicLink()) throw new Error('Expected regular file: ' + f);
  const stamp = [s.dev, s.ino, s.size, s.mtimeMs, s.ctimeMs].join(':');
  if (previous && previous !== stamp) throw new Error('Input changed: ' + f);
  trackedReads.set(f, stamp);
}

// Large reads and all SHA256 work are gated by main's explicit final completion.
async function facts(file, countLines = false) {
  remember(file);
  const hash = createHash('sha256');
  let bytes = 0, lines = 0;
  for await (const chunk of fs.createReadStream(file)) {
    hash.update(chunk); bytes += chunk.length;
    if (countLines) for (let i = chunk.indexOf(10); i >= 0; i = chunk.indexOf(10, i + 1)) lines++;
  }
  remember(file);
  return { sha256: hash.digest('hex'), bytes, lines };
}

function existingFiles(dir, prefix = '') {
  if (!fs.existsSync(dir)) return [];
  if (fs.lstatSync(dir).isSymbolicLink()) throw new Error('Archive symlink: ' + dir);
  return fs.readdirSync(dir, { withFileTypes: true }).flatMap(entry => {
    const name = prefix + entry.name, full = path.join(dir, entry.name);
    if (entry.isSymbolicLink()) throw new Error('Archive symlink: ' + full);
    if (entry.isDirectory()) return existingFiles(full, name + '/');
    if (!entry.isFile()) throw new Error('Unexpected archive entry: ' + full);
    return [name];
  });
}
