const borrowedCompareGroupDN = "cn=group," + smallIndexedPeopleDN
const borrowedCompareMemberDN = "uid=person-00000," + smallIndexedPeopleDN

func borrowedCompareGroup() directory.Entry {
	members := make([]string, 1000)
	for i := range members {
		members[i] = fmt.Sprintf("uid=person-%05d,%s", i, smallIndexedPeopleDN)
	}
	return nestGroupTestGroup(borrowedCompareGroupDN, members...)
}

func borrowedCompareResponse(t *testing.T, server *Server, state *connectionState, message ldapwire.Message) []byte {
	t.Helper()
	capture := &smallIndexedCapture{}
	if err := server.handleCompare(t.Context(), capture, state, message, message.Request.(ldapwire.CompareRequest)); err != nil {
		t.Fatal(err)
	}
	return bytes.Clone(capture.Bytes())
}

func TestCompareBorrowedDifferential(t *testing.T) {
	for _, backend := range []string{"bolt", "memory"} {
		t.Run(backend, func(t *testing.T) {
			server, state := newSmallIndexedFixture(t, 2)
			state.protocolVersion = 3
			if backend == "memory" {
				memory := storage.NewMemory()
				t.Cleanup(func() { _ = memory.Close() })
				server.config.Store = memory
			}
			group := borrowedCompareGroup()
			const referralDN = "cn=referral," + smallIndexedPeopleDN
			smallIndexedPut(t, server, state, group, false)
			smallIndexedPut(t, server, state, nestGroupTestEntry(referralDN, "referral", "ref", "ldap://example.test/dc=remote"), false)
			filter, err := ldap.CompileFilter("(cn=missing)")
			if err != nil {
				t.Fatal(err)
			}
			assertion := []ldapwire.Control{{OID: assertionControlOID, Critical: true, HasValue: true, Value: filter.Bytes()}}
			dn, err := state.runtime.schema.NormalizeDN(group.DN)
			if err != nil {
				t.Fatal(err)
			}
			database, store := databaseForNormalizedDN(state.runtime, dn), server.config.Store
			revision, known := server.currentStorageSnapshotRevision(t.Context())
			for _, test := range []struct {
				name, dn, attribute, value string
				controls                   []ldapwire.Control
				denied                     bool
				code                       ldapwire.ResultCode
			}{
				{name: "first", code: ldapwire.ResultCompareTrue},
				{name: "last", value: "uid=person-00999," + smallIndexedPeopleDN, code: ldapwire.ResultCompareTrue},
				{name: "normalized", value: "UID=PERSON-00000,OU=PEOPLE,DC=EXAMPLE,DC=COM", code: ldapwire.ResultCompareTrue},
				{name: "false", value: "uid=missing," + smallIndexedPeopleDN, code: ldapwire.ResultCompareFalse},
				{name: "missing attribute", attribute: "seeAlso", code: ldapwire.ResultNoSuchAttribute},
				{name: "missing entry", dn: "cn=missing," + smallIndexedPeopleDN, code: ldapwire.ResultNoSuchObject},
				{name: "ACL denial", denied: true, code: ldapwire.ResultInsufficientAccessRights},
				{name: "ACL before missing attribute", denied: true, attribute: "seeAlso", code: ldapwire.ResultInsufficientAccessRights},
				{name: "ACL before assertion", denied: true, controls: assertion, code: ldapwire.ResultInsufficientAccessRights},
				{name: "assertion before missing attribute", attribute: "seeAlso", controls: assertion, code: ldapwire.ResultAssertionFailed},
				{name: "referral", dn: referralDN, code: ldapwire.ResultReferral},
				{name: "referral before assertion", dn: referralDN, controls: assertion, code: ldapwire.ResultReferral},
				{name: "syntax before ACL", denied: true, value: "not-a-dn", code: ldapwire.ResultInvalidAttributeSyntax},
				{name: "unknown critical control", controls: []ldapwire.Control{{OID: "1.2.3.999", Critical: true}}, code: ldapwire.ResultUnavailableCriticalExtension},
			} {
				t.Run(test.name, func(t *testing.T) {
					t.Cleanup(func() { server.config.Store = store })
					request := ldapwire.CompareRequest{DN: borrowedCompareGroupDN, Attribute: "member", Assertion: []byte(borrowedCompareMemberDN)}
					if test.dn != "" {
						request.DN = test.dn
					}
					if test.attribute != "" {
						request.Attribute = test.attribute
					}
					if test.value != "" {
						request.Assertion = []byte(test.value)
					}
					state.boundDN = smallIndexedRootDN
					if test.denied {
						state.boundDN = borrowedCompareMemberDN
					}
					message := ldapwire.Message{ID: 1, Request: request, Controls: test.controls}
					if got := server.canBorrowCompareEntry(state, database, message, request); got != (backend == "bolt" && len(test.controls) == 0) {
						t.Fatalf("borrow eligibility = %t", got)
					}
					// Change only store admission; preserve ACLs and reader decorators.
					observed := &smallIndexedObservedStore{Store: store}
					server.config.Store = observed
					want := borrowedCompareResponse(t, server, state, message)
					server.config.Store = store
					for range 2 {
						got := borrowedCompareResponse(t, server, state, message)
						packet, err := ber.DecodePacketErr(got)
						if err != nil || len(packet.Children) != 2 || packet.Children[1].Tag != ldapwire.ApplicationCompareResponse || len(packet.Children[1].Children) < 3 {
							t.Fatalf("invalid Compare response: %v", err)
						}
						code, err := ber.ParseInt64(packet.Children[1].Children[0].Data.Bytes())
						if err != nil || code != int64(test.code) || !bytes.Equal(got, want) {
							t.Fatalf("Compare result = %d / %v, want %d; identical wire = %t", code, err, test.code, bytes.Equal(got, want))
						}
						if observed.updates != 0 {
							t.Fatal("fallback wrote storage")
						}
					}
				})
			}
			if after, available := server.currentStorageSnapshotRevision(t.Context()); !known || !available || after != revision {
				t.Fatal("Compare changed storage revision")
			}
			if err := store.View(t.Context(), func(reader storage.Reader) error {
				entry, err := readerForDatabase(reader, *database).Get(dn)
				if err == nil && !reflect.DeepEqual(entry.Attributes, group.Attributes) {
					t.Fatal("Compare mutated shared stored attributes")
				}
				return err
			}); err != nil {
				t.Fatal(err)
			}
		})
	}
}

func TestCompareBorrowedEligibility(t *testing.T) {
	server, state := newSmallIndexedFixture(t, 1)
	state.protocolVersion = 3
	database := databaseForDN(state.runtime, staticRuntimeDN(smallIndexedPeopleDN))
	for _, mode := range []string{"eligible", "v2", "controls", "empty controls", "ordinary attribute", "store", "normalizer", "other schema", "identity", "relay", "sync", "projection", "ACL callback"} {
		t.Run(mode, func(t *testing.T) {
			originalState, originalRuntime, originalDatabase, store := *state, *state.runtime, *database, server.config.Store
			t.Cleanup(func() {
				*state = originalState
				*state.runtime = originalRuntime
				*database = originalDatabase
				server.config.Store = store
			})
			request := ldapwire.CompareRequest{DN: borrowedCompareMemberDN, Attribute: "member", Assertion: []byte(borrowedCompareMemberDN)}
			message := ldapwire.Message{ID: 1, Request: request}
			custom := &defaultProjectionNormalizer{Registry: state.runtime.schema}
			switch mode {
			case "v2":
				state.protocolVersion = 2
			case "controls":
				message.Controls = []ldapwire.Control{{OID: "1.2.3.999"}}
			case "empty controls":
				message.ControlsPresent = true
			case "ordinary attribute":
				request.Attribute = "cn"
			case "store":
				server.config.Store = &smallIndexedObservedStore{Store: store}
			case "normalizer":
				database.dnNormalizer = custom
			case "other schema":
				database.dnNormalizer = state.runtime.schema.Clone()
			case "identity":
				database.dnNormalizer = nil
			case "relay":
				database.relay = &relayRuntimeConfiguration{}
			case "sync":
				database.syncProvider = true
			case "projection":
				state.runtime.features.retcode = true
			case "ACL callback":
				state.runtime.access = defaultProjectionPolicy(t, defaultProjectionRules(t, `to * by group.exact="cn=readers,dc=example,dc=com" read by * none`), nil)
			}
			if got := server.canBorrowCompareEntry(state, database, message, request); got != (mode == "eligible") || len(custom.trace) != 0 {
				t.Fatalf("eligibility = %t, normalizer callbacks = %v", got, custom.trace)
			}
			if observed, ok := server.config.Store.(*smallIndexedObservedStore); ok && (observed.views != 0 || observed.updates != 0) {
				t.Fatal("eligibility invoked an unknown store callback")
			}
			if mode == "normalizer" {
				server.config.Store = &smallIndexedObservedStore{Store: store}
				want := borrowedCompareResponse(t, server, state, message)
				trace := slices.Clone(custom.trace)
				server.config.Store, custom.trace = store, nil
				got := borrowedCompareResponse(t, server, state, message)
				if len(trace) == 0 || !slices.Equal(trace, custom.trace) || !bytes.Equal(want, got) {
					t.Fatal("fallback changed custom normalizer callbacks or errors")
				}
			}
		})
	}
}

func TestCompareBorrowedReaderContextAndNestedReads(t *testing.T) {
	server, state := newSmallIndexedFixture(t, 2)
	database := databaseForDN(state.runtime, staticRuntimeDN(smallIndexedPeopleDN))
	group := borrowedCompareGroup()
	smallIndexedPut(t, server, state, group, false)
	subject := acl.Subject{DN: smallIndexedRootDN, RealDN: borrowedCompareMemberDN, SSF: 128}
	ctx := withACLSubject(t.Context(), subject)
	stop := errors.New("callback result")
	if err := server.config.Store.View(ctx, func(reader storage.Reader) error {
		for _, wrapped := range []storage.Reader{
			accessContextReader{Reader: storageRevisionReader{Reader: reader, revision: 37}, subject: subject, ctx: ctx},
			storageRevisionReader{Reader: accessContextReader{Reader: reader, subject: subject, ctx: ctx}, revision: 41},
		} {
			revision, known := storage.ReaderSnapshotRevision(wrapped)
			handled, err := withCompareBorrowedReader(wrapped, func(borrowed storage.Reader) error {
				switch outer := borrowed.(type) {
				case accessContextReader:
					if outer.Reader.(storageRevisionReader).revision != 37 {
						t.Fatal("reader lost inner revision fallback")
					}
				case storageRevisionReader:
					if outer.revision != 41 {
						t.Fatal("reader lost outer revision fallback")
					}
				}
				if got, ok := readerACLSubject(borrowed); !ok || got != subject || borrowed.(interface{ StorageContext() context.Context }).StorageContext() != ctx {
					t.Fatal("reader lost ACL subject or operation context")
				}
				if got, ok := storage.ReaderSnapshotRevision(borrowed); !known || !ok || got != revision {
					t.Fatal("reader lost snapshot revision")
				}
				tx := readerForDatabase(borrowed, *database)
				read := func(tx storage.Reader, raw string) directory.Entry {
					dn, err := state.runtime.schema.NormalizeDN(raw)
					if err != nil {
						t.Fatal(err)
					}
					entry, err := tx.Get(dn)
					if err != nil {
						t.Fatal(err)
					}
					return entry
				}
				first := read(tx, group.DN)
				want := first.Clone()
				other := read(tx, borrowedCompareMemberDN)
				owned := other.Clone()
				clear(other.Attributes[0].Values[0])
				if nested, err := withCompareBorrowedReader(wrapped, func(inner storage.Reader) error {
					read(readerForDatabase(inner, *database), borrowedCompareMemberDN)
					return nil
				}); !nested || err != nil {
					t.Fatalf("nested borrow = %t / %v", nested, err)
				}
				if !reflect.DeepEqual(first, want) || !reflect.DeepEqual(read(tx, borrowedCompareMemberDN), owned) {
					t.Fatal("subsequent Get or nested decoder reuse changed retained or stored entry")
				}
				return stop
			})
			if !handled || !errors.Is(err, stop) {
				t.Fatalf("borrow callback result = %t / %v", handled, err)
			}
			custom := &defaultProjectionContextReader{Reader: wrapped}
			if handled, err := withCompareBorrowedReader(storageRevisionReader{Reader: custom}, func(storage.Reader) error {
				t.Fatal("unknown reader invoked borrow callback")
				return nil
			}); handled || err != nil || custom.calls != 0 {
				t.Fatal("unknown reader was admitted or called speculatively")
			}
		}
		return nil
	}); err != nil {
		t.Fatal(err)
	}
}
