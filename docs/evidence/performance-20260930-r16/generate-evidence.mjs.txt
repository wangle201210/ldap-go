}

const digest = content => createHash('sha256').update(content).digest('hex');
async function sha(f) {
  const stat = regularFile(f), hash = createHash('sha256');
  for await (const chunk of fs.createReadStream(f)) hash.update(chunk);
  remember(f, stat);
  return hash.digest('hex');
}

// Refuse stale or foreign files rather than silently excluding them from SHA256SUMS.
function existingFiles(dir, prefix = '') {
  if (!fs.existsSync(dir)) return [];
  if (fs.lstatSync(dir).isSymbolicLink()) throw new Error(`Refusing archive directory symlink: ${dir}`);
  return fs.readdirSync(dir, { withFileTypes: true }).flatMap(entry => {
    const name = prefix + entry.name, full = path.join(dir, entry.name);
    if (entry.isSymbolicLink()) throw new Error(`Refusing archive symlink: ${full}`);
    if (entry.isDirectory()) return existingFiles(full, name + '/');
    if (!entry.isFile()) throw new Error(`Unexpected archive entry: ${full}`);
    return [name];
  });
}
