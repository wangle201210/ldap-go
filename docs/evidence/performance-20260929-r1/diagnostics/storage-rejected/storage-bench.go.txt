package storage

import (
	"fmt"
	"testing"
)

// Isolate validation from decoding and callbacks, including the small posting
// sets used by bounded member searches.
func BenchmarkBoltEqualityCandidatePrevalidation(b *testing.B) {
	store, _, _ := newBoltCandidateStore(b, minEncodedEqualityIndexCandidates)
	if err := store.View(b.Context(), func(reader Reader) error {
		tx := reader.(*boltTx)
		references := boltCandidateReferences(b, tx)
		for _, count := range []int{1, 4, minEncodedEqualityIndexCandidates} {
			b.Run(fmt.Sprintf("N%d", count), func(b *testing.B) {
				b.ReportAllocs()
				for b.Loop() {
					candidates, valid, err := tx.prevalidateEqualityIndexCandidates("db", references[:count])
					if err != nil || !valid || len(candidates) != count {
						b.Fatalf("candidate count/valid/error = %d/%v/%v", len(candidates), valid, err)
					}
				}
			})
		}
		return nil
	}); err != nil {
		b.Fatal(err)
	}
}
