#!/bin/sh
set -eu
. /tmp/ldap-go-compatibility-audit-20260920.env
audit=/var/tmp/ldap-go-perf-round16-20260923
out=$audit/concurrent-recheck
mkdir "$out"
umask 077
printf '%s' scale-qualification-local-secret >"$out/password"
awk 'BEGIN {for(i=1001;i<=2000;i++) printf "scale-%06d\n",i}' >"$out/values.txt"
printf 'run\trepeat\tms\n' >"$out/timings.tsv"
pid=
stop_server() {
  if [ -n "$pid" ]; then
    kill -TERM "$pid" 2>/dev/null || true
    wait "$pid" 2>/dev/null || true
    pid=
  fi
}
trap 'stop_server; rm "$out/password"' EXIT HUP INT TERM
root_dn=cn=admin,dc=scale,dc=qualification
uri=ldap://127.0.0.1:29481
for run in current-1 before-1 before-2 current-2 current-3 before-3; do
  side=${run%-*}
  directory=$out/$run
  mkdir "$directory"
  cp /var/tmp/ldap-go-perf-round4-20260922/online-accepted/current-1/data.db "$directory/data.db"
  LDAP_GO_ROOT_PASSWORD=scale-qualification-local-secret "$audit/$side" serve \
    -db "$directory/data.db" -listen 127.0.0.1:29481 -root-dn "$root_dn" \
    -search-limit 100100 -search-candidate-limit 100100 \
    -search-candidate-bytes 819200000 -search-memory-bytes 1638400000 \
    >"$directory/server.log" 2>&1 &
  pid=$!
  deadline=$(( $(date +%s) + 60 ))
  until ldapwhoami -H "$uri" -x -D "$root_dn" -y "$out/password" -o nettimeout=1 >/dev/null 2>&1; do
    kill -0 "$pid"
    [ "$(date +%s)" -lt "$deadline" ]
    sleep 0.05
  done
  LDAP_BENCH_PASSWORD=scale-qualification-local-secret /var/tmp/ldap-go-perf-round4-20260922/ldapbench \
    -uri "$uri" -bind-dn "$root_dn" -password-env LDAP_BENCH_PASSWORD \
    -base dc=scale,dc=qualification -people ou=people,dc=scale,dc=qualification \
    -read-only -n 2 -entries 100000 -label "$run-warmup" >"$directory/warmup.json"
  for repeat in 0 1 2 3; do
    start=$(gdate +%s%N)
    workers=
    for worker in 0 1 2 3 4 5 6 7; do
      ldapsearch -H "$uri" -x -D "$root_dn" -y "$out/password" -LLL -o ldif-wrap=no \
        -b ou=people,dc=scale,dc=qualification -f "$out/values.txt" '(uid=%s)' uid >"$directory/client-$worker.ldif" &
      workers="$workers $!"
    done
    for worker_pid in $workers; do wait "$worker_pid"; done
    end=$(gdate +%s%N)
    printf '%s\t%s\t%s\n' "$run" "$repeat" "$(((end-start)/1000000))" >>"$out/timings.tsv"
    for worker in 0 1 2 3 4 5 6 7; do
      [ "$(awk '/^dn: / {n++} END {print n+0}' "$directory/client-$worker.ldif")" -eq 1000 ]
      awk '/^uid: / {print $2}' "$directory/client-$worker.ldif" >"$directory/client-values.txt"
      cmp "$out/values.txt" "$directory/client-values.txt"
    done
  done
  stop_server
  rm "$directory/data.db"
  printf 'Concurrent recheck passed: %s\n' "$run"
done
