#!/bin/sh
set -eu
. /tmp/ldap-go-compatibility-audit-20260920.env
audit=/var/tmp/ldap-go-perf-round14-20260923
source_dir=/var/tmp/ldap-go-perf-round4-20260922/online-accepted
out=$audit/interleaved
mkdir "$out"
umask 077
printf '%s' scale-qualification-local-secret >"$out/password"
pids=
cleanup() {
  for pid in $pids; do kill -TERM "$pid" 2>/dev/null || true; done
  for pid in $pids; do wait "$pid" 2>/dev/null || true; done
  pids=
  rm "$out/password"
}
trap cleanup EXIT HUP INT TERM
root_dn=cn=admin,dc=scale,dc=qualification
people_dn=ou=people,dc=scale,dc=qualification
for side in before current openldap; do
  case "$side" in before) port=29481;; current) port=29482;; openldap) port=29483;; esac
  uri=ldap://127.0.0.1:$port
  directory=$out/$side
  mkdir "$directory"
  if [ "$side" = openldap ]; then
    mkdir "$directory/data"
    cp "$source_dir/openldap-1/data/data.mdb" "$directory/data/data.mdb"
    sed -e "s|^directory .*|directory $directory/data|" \
      -e "s|^pidfile .*|pidfile $directory/slapd.pid|" \
      -e "s|^argsfile .*|argsfile $directory/slapd.args|" \
      -e "s|^include .*/core.schema|include $OPENLDAP_SCHEMA_DIR/core.schema|" \
      -e "s|^include .*/cosine.schema|include $OPENLDAP_SCHEMA_DIR/cosine.schema|" \
      -e "s|^include .*/inetorgperson.schema|include $OPENLDAP_SCHEMA_DIR/inetorgperson.schema|" \
      "$source_dir/openldap-1/slapd.conf" >"$directory/slapd.conf"
    slapd -f "$directory/slapd.conf" -h "$uri" -d 0 >"$directory/server.log" 2>&1 &
  else
    cp "$source_dir/current-1/data.db" "$directory/data.db"
    LDAP_GO_ROOT_PASSWORD=scale-qualification-local-secret "$audit/$side" serve \
      -db "$directory/data.db" -listen 127.0.0.1:$port -root-dn "$root_dn" \
      -search-limit 100100 -search-candidate-limit 100100 \
      -search-candidate-bytes 819200000 -search-memory-bytes 1638400000 \
      >"$directory/server.log" 2>&1 &
  fi
  pid=$!
  pids="$pids $pid"
  deadline=$(( $(date +%s) + 60 ))
  until ldapwhoami -H "$uri" -x -D "$root_dn" -y "$out/password" -o nettimeout=1 >/dev/null 2>&1; do
    kill -0 "$pid"
    [ "$(date +%s)" -lt "$deadline" ]
    sleep 0.05
  done
  LDAP_BENCH_PASSWORD=scale-qualification-local-secret /var/tmp/ldap-go-perf-round4-20260922/ldapbench \
    -uri "$uri" -bind-dn "$root_dn" -password-env LDAP_BENCH_PASSWORD \
    -base dc=scale,dc=qualification -people "$people_dn" -read-only -n 2 -entries 100000 \
    -label "$side-warmup" >"$directory/warmup.json"
done
for round in 1 2 3; do
  case "$round" in 1) order='before current openldap';; 2) order='current openldap before';; 3) order='openldap before current';; esac
  for workload in user-bind long-dn long-password fast; do
    for side in $order; do
      case "$side" in before) port=29481;; current) port=29482;; openldap) port=29483;; esac
      uri=ldap://127.0.0.1:$port
      run_dir=$out/$side-$round
      mkdir -p "$run_dir"
      printf 'Recheck %s %s %s\n' "$round" "$workload" "$side"
      case "$workload" in
        user-bind)
          LDAP_BENCH_PASSWORD=scale-qualification-local-secret "$audit/user-probe" -uri "$uri" -n 3000 >"$run_dir/user-bind.json";;
        long-dn|long-password)
          LDAP_BENCH_PASSWORD=scale-qualification-local-secret "$audit/extended-probe" -uri "$uri" -n 3000 -"$workload" >"$run_dir/$workload.json";;
        fast)
          for repeat in 1 2 3; do
            LDAP_BENCH_PASSWORD=scale-qualification-local-secret /var/tmp/ldap-go-perf-round6-20260923/fast-probe \
              -uri "$uri" -bind-dn "$root_dn" -password-env LDAP_BENCH_PASSWORD \
              -base dc=scale,dc=qualification -people "$people_dn" -read-only -n 3000 -entries 100000 \
              -label "$side-$round-$repeat" >"$run_dir/fast-$repeat.json"
          done;;
      esac
    done
  done
done
printf 'run\tcanonical_entries\tcanonical_cksum\tcanonical_bytes\n' >"$out/validation.tsv"
for side in before current openldap; do
  case "$side" in before) port=29481;; current) port=29482;; openldap) port=29483;; esac
  directory=$out/$side
  ldapsearch -H ldap://127.0.0.1:$port -x -D "$root_dn" -y "$out/password" \
    -LLL -o ldif-wrap=no -E pr=10000/noprompt -b dc=scale,dc=qualification '(objectClass=*)' '*' >"$directory/data.ldif"
  /var/tmp/ldap-go-perf-round2-20260922/ldifcanonical -in "$directory/data.ldif" >"$directory/data.unsorted"
  LC_ALL=C sort "$directory/data.unsorted" >"$directory/data.canonical"
  rm "$directory/data.unsorted"
  count=$(wc -l <"$directory/data.canonical" | tr -d ' ')
  [ "$count" -eq 100002 ]
  set -- $(cksum "$directory/data.canonical")
  [ "$1" = 2143929969 ]
  [ "$2" = 42712438 ]
  printf '%s\t%s\t%s\t%s\n' "$side" "$count" "$1" "$2" >>"$out/validation.tsv"
  cmp "$audit/online/before-1/data.canonical" "$directory/data.canonical"
done
cleanup
trap - EXIT HUP INT TERM
rm "$out/before/data.db" "$out/current/data.db" "$out/openldap/data/data.mdb"
printf 'Interleaved recheck passed.\n'
